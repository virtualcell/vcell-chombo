# vcell-chombo builds on Windows through MSYS2/MinGW-w64, not MSVC.
#
# Chombo's build system is GNU make plus a perl preprocessor (ChomboFortran)
# driven through a Unix shell, and there is no reimplementation of that behind
# MSVC. MinGW-w64 supplies all of it while still producing native Windows
# binaries with no runtime dependency on an emulation DLL. The historical VCell
# build used Cygwin for the same reasons, and Chombo's makefiles are folded onto
# that same CYGWIN branch here -- which is also what yields -DCH_CYGWIN, the
# macro VCellChombo's SimTool tests to select the Windows path separator.
#
# Messaging is never built on Windows, so libcurl and openssl never enter the
# dependency graph at all; conanfile.py drops the requirement outright rather
# than leaving it to the option.
#
# The version is pinned to what MSYS2 currently ships, which is well ahead of
# the GCC 13 the Linux and macOS profiles use. That is not a preference: MSYS2
# offers one GCC, and holding it back would mean pinning packages by hand.
[settings]
arch=x86_64
build_type=Release
compiler=gcc
compiler.version=16
compiler.libcxx=libstdc++11
compiler.cppstd=20
compiler.threads=posix
compiler.exception=seh
os=Windows

[conf]
# Required: conanfile.py's layout() only flattens the tree to build/generators
# when this is set. Without it, cmake_layout() emits build/Release/generators.
tools.cmake.cmaketoolchain:generator=Ninja

# HDF5 <= 1.14.6 does not compile under this toolchain without this. GCC 16
# advertises _Float16, so HDF5's ConfigureChecks turns on its FLOAT16 conversion
# paths, but mingw-w64's <float.h> defines no FLT16_MAX and H5Tconv_float.c
# fails with "'FLT16_MAX' undeclared". The Conan recipe exposes no option for
# it, so the CMake variable is injected into that dependency's build directly.
#
# Nothing here uses _Float16 -- the solver is entirely double precision -- so
# this removes a capability we would never call rather than working around a
# defect in something we depend on.
hdf5/*:tools.cmake.cmaketoolchain:extra_variables={'HDF5_ENABLE_NONSTANDARD_FEATURE_FLOAT16': 'OFF'}

# xz_utils refuses to configure here otherwise. Its CMake finds MSYS2's gettext
# 0.22.5, and native language support needs UCRT with gettext-runtime >= 0.23.1,
# so it stops with "set XZ_NLS=OFF" -- which is upstream's own sanctioned escape
# hatch rather than something invented here. NLS only supplies translated error
# messages in a library we link for its compression, so nothing is lost.
#
# xz arrives transitively through libzip. It could be dropped from the graph
# altogether by turning off libzip's LZMA backend, but that would narrow which
# archives the solver can read, on every platform, to save a source build on one.
xz_utils/*:tools.cmake.cmaketoolchain:extra_variables={'XZ_NLS': 'OFF'}

[options]
fPIC=True
shared=False
